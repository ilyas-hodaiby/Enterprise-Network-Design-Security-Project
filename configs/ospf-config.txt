! ===== OSPF Configuration =====
! Dynamic routing across all floors
! All subnets in Area 0 (backbone)

router ospf 1
 router-id 2.2.2.2

 ! Advertise all department subnets
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.40.0 0.0.0.31 area 0
 network 192.168.50.0 0.0.0.127 area 0

 ! Uplink to edge router
 network 10.0.0.0 0.0.0.3 area 0

 ! Stop OSPF hello packets on user-facing ports
 ! Prevents OSPF manipulation from end devices
 passive-interface default
 no passive-interface GigabitEthernet0/0

 ! Advertise default route to all OSPF neighbours
 default-information originate

! ===== OSPF Verification Commands =====
! show ip ospf neighbor
! show ip route ospf
! show ip ospf interface
